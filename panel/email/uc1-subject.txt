function brl(v) {
    var n = Number(v);
    if (!n && n !== 0) { return ''; }
    var neg = n < 0;
    n = Math.abs(n);
    var whole = Math.floor(n);
    var cents = Math.round((n - whole) * 100);
    if (cents === 100) { whole += 1; cents = 0; }
    var w = String(whole);
    var out = '';
    for (var k = 0; k < w.length; k++) {
        if (k > 0 && (w.length - k) % 3 === 0) { out += '.'; }
        out += w.charAt(k);
    }
    return (neg ? '-' : '') + out + ',' + (cents < 10 ? '0' + cents : String(cents));
}
